   912→function SniperCalibrator({ imgSrc, imgW, imgH, landmark, point, onConfirm, goPrev, goNext, progressLabel, isConfirmed, onSetPoint }: {
   913→  imgSrc: string;
   914→  imgW: number;
   915→  imgH: number;
   916→  landmark: LandmarkDef | undefined;
   917→  point: CalibratedPoint | undefined;
   918→  onConfirm: (pt: Point) => void;
   919→  goPrev: () => void;
   920→  goNext: () => void;
   921→  progressLabel: string;
   922→  isConfirmed: boolean;
   923→  onSetPoint: (pt: Point) => void;
   924→}) {
   925→  const viewportRef = useRef<HTMLDivElement>(null);
   926→  const [zoom, setZoom] = useState(2.35);
   927→  const [pan, setPan] = useState({ x: 0, y: 0 });
   928→  const [dragging, setDragging] = useState(false);
   929→  const dragRef = useRef({ x: 0, y: 0, panX: 0, panY: 0 });
   930→
   931→  // When landmark changes, fly to it
   932→  useEffect(() => {
   933→    if (!point) return;
   934→    // center the landmark under the crosshair
   935→    const targetPanX = -(point.x - imgW / 2) * zoom;
   936→    const targetPanY = -(point.y - imgH / 2) * zoom;
   937→    // smooth snap
   938→    setPan({ x: targetPanX, y: targetPanY });
   939→    // eslint-disable-next-line react-hooks/exhaustive-deps
   940→  }, [landmark?.id]);
   941→
   942→  // Keep pan updated when zoom changes: keep crosshair world stable
   943→  const setZoomKeepCenter = (nz: number) => {
   944→    const ratio = nz / zoom;
   945→    setPan(p => ({ x: p.x * ratio, y: p.y * ratio }));
   946→    setZoom(nz);
   947→  };
   948→
   949→  const crossWorld = useMemo(() => {
   950→    // image coordinate under crosshair
   951→    return {
   952→      x: imgW / 2 + (-pan.x) / zoom,
   953→      y: imgH / 2 + (-pan.y) / zoom,
   954→    };
   955→  }, [pan, zoom, imgW, imgH]);
   956→
   957→  // update the live point as user pans (for magnifier accuracy)
   958→  useEffect(() => {
   959→    if (!point) return;
   960→    onSetPoint(crossWorld);
   961→    // eslint-disable-next-line react-hooks/exhaustive-deps
   962→  }, [crossWorld.x, crossWorld.y]);
   963→
   964→  const onWheel = (e: React.WheelEvent) => {
   965→    e.preventDefault();
   966→    const delta = -e.deltaY;
   967→    const factor = delta > 0 ? 1.12 : 0.89;
   968→    const nz = Math.min(6.5, Math.max(0.7, zoom * factor));
   969→    setZoomKeepCenter(nz);
   970→  };
   971→
   972→  const onPointerDown = (e: React.PointerEvent) => {
   973→    (e.currentTarget as Element).setPointerCapture(e.pointerId);
   974→    setDragging(true);
   975→    dragRef.current = { x: e.clientX, y: e.clientY, panX: pan.x, panY: pan.y };
   976→  };
   977→  const onPointerMove = (e: React.PointerEvent) => {
   978→    if (!dragging) return;
   979→    const dx = e.clientX - dragRef.current.x;
   980→    const dy = e.clientY - dragRef.current.y;
   981→    setPan({ x: dragRef.current.panX + dx, y: dragRef.current.panY + dy });
   982→  };
   983→  const onPointerUp = () => setDragging(false);
   984→
   985→  // keyboard nudging - move the image under the crosshair for sub-pixel precision
   986→  useEffect(() => {
   987→    const h = (e: KeyboardEvent) => {
   988→      const step = e.shiftKey ? 0.45 : 1.65;
   989→      let dx = 0, dy = 0;
   990→      if (e.key === "ArrowLeft") { dx = -step; e.preventDefault(); }
   991→      if (e.key === "ArrowRight") { dx = step; e.preventDefault(); }
   992→      if (e.key === "ArrowUp") { dy = -step; e.preventDefault(); }
   993→      if (e.key === "ArrowDown") { dy = step; e.preventDefault(); }
   994→      if (dx !== 0 || dy !== 0) {
   995→        // move image opposite so the point under crosshair moves with arrow direction
   996→        setPan(p => ({ x: p.x - dx * zoom, y: p.y - dy * zoom }));
   997→      }
   998→      if (e.key === "Enter") { if (point) onConfirm({ x: crossWorld.x, y: crossWorld.y }); }
   999→    };
  1000→    window.addEventListener("keydown", h);
  1001→    return () => window.removeEventListener("keydown", h);
  1002→  }, [zoom, onConfirm, point, crossWorld]);
  1003→
  1004→  // Magnifier canvas
  1005→  const magRef = useRef<HTMLCanvasElement>(null);
  1006→  const imgRef = useRef<HTMLImageElement>(null);
  1007→
  1008→  useEffect(() => {
  1009→    const canvas = magRef.current;
  1010→    const img = imgRef.current;
  1011→    if (!canvas || !img || !img.complete) return;
  1012→    const ctx = canvas.getContext("2d");
  1013→    if (!ctx) return;
  1014→    const size = 168;
  1015→    canvas.width = size;
  1016→    canvas.height = size;
  1017→    ctx.imageSmoothingEnabled = false;
  1018→    const srcSize = 72 / zoom; // world size visible
  1019→    const sx = Math.max(0, Math.min(imgW - srcSize, crossWorld.x - srcSize / 2));
  1020→    const sy = Math.max(0, Math.min(imgH - srcSize, crossWorld.y - srcSize / 2));
  1021→    ctx.clearRect(0, 0, size, size);
  1022→    ctx.drawImage(img, sx, sy, srcSize, srcSize, 0, 0, size, size);
  1023→    // crosshair
  1024→    ctx.strokeStyle = "rgba(255,255,255,0.9)";
  1025→    ctx.lineWidth = 1;
  1026→    ctx.beginPath();
  1027→    ctx.moveTo(size / 2, 8); ctx.lineTo(size / 2, size - 8);
  1028→    ctx.moveTo(8, size / 2); ctx.lineTo(size - 8, size / 2);
  1029→    ctx.stroke();
  1030→    ctx.strokeStyle = "#06c1e4";
  1031→    ctx.beginPath();
  1032→    ctx.arc(size / 2, size / 2, 3.2, 0, Math.PI * 2);
  1033→    ctx.stroke();
  1034→  }, [crossWorld, imgW, imgH, zoom, imgSrc]);
  1035→
  1036→  if (!landmark) return null;
  1037→
  1038→  return (
  1039→    <div className="bg-white rounded-[24px] shadow-[0_14px_60px_rgba(15,15,20,0.10)] ring-1 ring-zinc-200 overflow-hidden">
  1040→      <div className="px-5 sm:px-7 pt-5 pb-4 border-b border-zinc-100 flex flex-wrap items-start justify-between gap-3">
  1041→        <div>
  1042→          <div className="text-[11.7px] text-zinc-500">Align to crosshair • {progressLabel}</div>
  1043→          <div className="text-[24px] font-[650] tracking-tight">{landmark.name}</div>
  1044→          <div className="text-[13.3px] text-zinc-500">{landmark.hint}</div>
  1045→        </div>
  1046→        <div className="flex items-center gap-2">
  1047→          <button onClick={goPrev} className="px-3 py-1.5 rounded-full border border-zinc-300 text-[12.8px] text-zinc-700 hover:bg-zinc-50">Prev</button>
  1048→          <button onClick={goNext} className="px-3 py-1.5 rounded-full border border-zinc-300 text-[12.8px] text-zinc-700 hover:bg-zinc-50">Next</button>
  1049→          <button
  1050→            onClick={() => onConfirm({ x: crossWorld.x, y: crossWorld.y })}
  1051→            className={`px-4 py-1.5 rounded-full text-[13px] font-medium text-white ${isConfirmed ? "bg-emerald-600 hover:bg-emerald-600/90" : "bg-zinc-900 hover:bg-zinc-800"}`}
  1052→          >
  1053→            {isConfirmed ? "Confirmed ✓" : "Confirm / Next"}
  1054→          </button>
  1055→        </div>
  1056→      </div>
  1057→
  1058→      <div className="relative">
  1059→        <div
  1060→          ref={viewportRef}
  1061→          onWheel={onWheel}
  1062→          onPointerDown={onPointerDown}
  1063→          onPointerMove={onPointerMove}
  1064→          onPointerUp={onPointerUp}
  1065→          className="relative w-full h-[600px] md:h-[660px] bg-[#eef0f3] overflow-hidden touch-none select-none cursor-grab active:cursor-grabbing"
  1066→        >
  1067→          {/* The image, panned */}
  1068→          <img
  1069→            ref={imgRef}
  1070→            src={imgSrc}
  1071→            alt=""
  1072→            draggable={false}
  1073→            className="absolute left-1/2 top-1/2 max-w-none pointer-events-none"
  1074→            style={{
  1075→              width: imgW,
  1076→              height: imgH,
  1077→              transform: `translate(calc(-50% + ${pan.x}px), calc(-50% + ${pan.y}px)) scale(${zoom})`,
  1078→              transformOrigin: "center center",
  1079→            }}
  1080→          />
  1081→
  1082→          {/* Crosshair center */}
  1083→          <div className="pointer-events-none absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2">
  1084→            {/* Outer rings */}
  1085→            <div className="w-[66px] h-[66px] rounded-full border border-zinc-900/16"></div>
  1086→            <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-[32px] h-[32px] rounded-full border border-zinc-900/24"></div>
  1087→            {/* cross lines */}
  1088→            <div className="absolute left-1/2 top-1/2 w-[150px] h-px bg-zinc-900/55 -translate-x-1/2 -translate-y-1/2"></div>
  1089→            <div className="absolute left-1/2 top-1/2 w-px h-[150px] bg-zinc-900/55 -translate-x-1/2 -translate-y-1/2"></div>
  1090→            {/* center dot */}
  1091→            <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-[7px] h-[7px] rounded-full bg-[#06c1e4] ring-4 ring-[#06c1e4]/20"></div>
  1092→          </div>
  1093→
  1094→          {/* Zoom / magnifier */}
  1095→          <div className="absolute right-4 top-4 bg-white/95 rounded-[18px] shadow-xl ring-1 ring-zinc-200 p-[11px]">
  1096→            <div className="text-[10.7px] text-zinc-500 mb-1.5 font-mono">4× MAGNIFIER • {zoom.toFixed(2)}×</div>

  1103→            Drag to pan • Scroll to zoom • Arrows nudge • Enter confirm
  1104→          </div>
  1105→        </div>
  1106→      </div>
  1107→    </div>
  1108→  );

  1126→  useEffect(() => { setImgDisplay({ w: fit.w, h: fit.h }); }, [fit]);

  1157→      const cat = r.metric.category;

  1175→      </div>
  1176→
  1177→      <div className="grid grid-cols-1 xl:grid-cols-[560px_1fr] gap-8 items-start">
  1178→        {/* Left Image */}
  1179→        <div className="bg-white rounded-[26px] shadow-[0_18px_70px_rgba(23,23,27,0.09)] ring-1 ring-zinc-200 p-5">
  1180→          <div className="relative mx-auto" style={{ width: imgDisplay.w, height: imgDisplay.h }}>
  1181→            <img

  1183→              src={imgSrc}

  1187→              className="rounded-[16px] block object-contain select-none"
